2. Create `extension/components/storage.ts` -- chrome.storage wrapper with Zod validation:
   ```typescript
   import browser from 'webextension-polyfill';
   import { StorageStateSchema, type StorageState, type CaptureEntry } from './types';

   /**
    * Load and validate storage state from chrome.storage.local.
    * Returns defaults for any missing fields (Zod defaults).
    */
   export async function loadStorage(): Promise<StorageState> {
     const raw = await browser.storage.local.get(null);
     return StorageStateSchema.parse(raw);
   }

   /**
    * Save partial storage state to chrome.storage.local.
    */
   export async function saveStorage(data: Partial<StorageState>): Promise<void> {
     await browser.storage.local.set(data);
   }

   /**
    * Get today's date key in YYYY-MM-DD format.
    */
   export function getToday(): string {
     return new Date().toISOString().split('T')[0];
   }

   /**
    * Save a capture entry for today, with deduplication by URL (D-05).
    * Returns true if the entry was saved (new), false if duplicate.
    */
   export async function saveCapture(entry: CaptureEntry): Promise<boolean> {
     const today = getToday();
     const { captures = {} } = await browser.storage.local.get('captures');

     if (!captures[today]) captures[today] = [];

     // D-05: Deduplicate per day -- first visit timestamp kept
     const exists = captures[today].some((c: CaptureEntry) => c.url === entry.url);
     if (exists) return false;

     captures[today].push(entry);
     await browser.storage.local.set({
       captures,
       lastCaptureTimestamp: entry.timestamp,
     });
     return true;
   }

   /**
    * Get the count of captures for today.
    */
   export async function getTodayCount(): Promise<number> {
     const today = getToday();
     const { captures = {} } = await browser.storage.local.get('captures');
     return (captures[today] || []).length;
   }

   /**
    * Get the last capture entry for today, or null if none.
    */
   export async function getLastCapture(): Promise<CaptureEntry | null> {
     const today = getToday();
     const { captures = {} } = await browser.storage.local.get('captures');
     const todayCaptures = captures[today] || [];
     return todayCaptures.length > 0 ? todayCaptures[todayCaptures.length - 1] : null;
   }
   ```

3. Create `extension/components/blocklist.ts` -- domain matching with subdomain support:
   ```typescript
   import browser from 'webextension-polyfill';
   import { BlocklistConfigSchema, type BlocklistConfig } from './types';

   /**
    * Check if hostname matches blocklist with subdomain matching (D-07).
    * Blocking "google.com" also blocks "mail.google.com", "docs.google.com", etc.
    */
   export function isBlocked(hostname: string, flatBlocklist: string[]): boolean {
     const normalized = hostname.toLowerCase();

     for (const pattern of flatBlocklist) {
       if (normalized === pattern) return true;
       if (normalized.endsWith('.' + pattern)) return true;
     }

     return false;
   }

   /**
    * Load blocklist from chrome.storage.local.
    * Falls back to default blocklist on first run or parse error.
    */
   export async function loadBlocklist(): Promise<string[]> {
     const { blocklist } = await browser.storage.local.get('blocklist');
     if (Array.isArray(blocklist) && blocklist.length > 0) {
       return blocklist;
     }
     // First run -- load defaults and persist
     const defaults = await loadDefaultBlocklist();
     const flat = flattenBlocklist(defaults);
     await browser.storage.local.set({ blocklist: flat });
     return flat;
   }

   /**
    * Load default blocklist from bundled blocklist.json.
    */
   export async function loadDefaultBlocklist(): Promise<BlocklistConfig> {
     const response = await fetch(browser.runtime.getURL('blocklist.json'));
     const json = await response.json();
     return BlocklistConfigSchema.parse(json);
   }

   /**
    * Flatten BlocklistConfig categories into a single string array.
    */
   export function flattenBlocklist(config: BlocklistConfig): string[] {
     return [
       ...config.google,
       ...config.banking,
       ...config.social,
       ...config.auth,
       ...(config.custom || []),
     ];
   }

   /**
    * Add a domain to the blocklist (D-03 quick-block, D-09 config file).
    * Persists to chrome.storage.local.
    */
   export async function addToBlocklist(domain: string): Promise<boolean> {
     const normalized = domain.toLowerCase();
     const { blocklist = [] } = await browser.storage.local.get('blocklist');

     if (blocklist.includes(normalized)) return false;

     blocklist.push(normalized);
     await browser.storage.local.set({ blocklist });
     return true;
   }

   /**
    * Remove a domain from the blocklist.
    */
   export async function removeFromBlocklist(domain: string): Promise<boolean> {
     const normalized = domain.toLowerCase();
     const { blocklist = [] } = await browser.storage.local.get('blocklist');

     const index = blocklist.indexOf(normalized);
     if (index === -1) return false;

     blocklist.splice(index, 1);
     await browser.storage.local.set({ blocklist });
     return true;
   }
   ```
