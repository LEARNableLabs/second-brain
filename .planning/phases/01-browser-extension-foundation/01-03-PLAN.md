**3. Replace `extension/entrypoints/popup/main.ts`:**

```typescript
import browser from 'webextension-polyfill';
import { loadStorage, saveStorage, getTodayCount, getLastCapture } from '../../components/storage';
import { addToBlocklist } from '../../components/blocklist';

// === DOM Elements ===
const pauseToggle = document.getElementById('pause-toggle') as HTMLInputElement;
const toggleLabel = document.getElementById('toggle-label') as HTMLLabelElement;
const pauseBanner = document.getElementById('pause-banner') as HTMLDivElement;
const countToday = document.getElementById('count-today') as HTMLDivElement;
const lastCaptureEl = document.getElementById('last-capture') as HTMLDivElement;
const statsSection = document.getElementById('stats-section') as HTMLDivElement;
const emptyState = document.getElementById('empty-state') as HTMLDivElement;
const blockBtn = document.getElementById('block-site') as HTMLButtonElement;
const editLink = document.getElementById('edit-blocklist') as HTMLAnchorElement;
const toast = document.getElementById('toast') as HTMLDivElement;

// === Initialize popup state from chrome.storage ===
async function initPopup(): Promise<void> {
  // Load pause state
  const storage = await loadStorage();
  pauseToggle.checked = storage.isPaused;
  updatePauseUI(storage.isPaused);

  // Load today's stats
  const count = await getTodayCount();
  const lastCapture = await getLastCapture();

  if (count === 0) {
    statsSection.style.display = 'none';
    emptyState.style.display = 'block';
  } else {
    statsSection.style.display = 'flex';
    emptyState.style.display = 'none';
    countToday.textContent = String(count);

    if (lastCapture) {
      const elapsed = Date.now() - lastCapture.timestamp;
      lastCaptureEl.textContent = formatTimeAgo(elapsed);
    }
  }

  // Update block button with current tab domain
  const [tab] = await browser.tabs.query({ active: true, currentWindow: true });
  if (tab?.url) {
    try {
      const hostname = new URL(tab.url).hostname;
      blockBtn.textContent = `Block this site: ${hostname}`;
      blockBtn.dataset.domain = hostname;
    } catch {
      blockBtn.disabled = true;
      blockBtn.textContent = 'Block this site';
    }
  } else {
    blockBtn.disabled = true;
  }
}

// === Pause Toggle (D-01, D-02) ===
pauseToggle.addEventListener('change', async () => {
  const isPaused = pauseToggle.checked;
  await saveStorage({ isPaused });
  updatePauseUI(isPaused);
});

function updatePauseUI(isPaused: boolean): void {
  if (isPaused) {
    toggleLabel.textContent = 'Resume capture';
    pauseBanner.classList.add('visible');
    statsSection.classList.add('dimmed');
  } else {
    toggleLabel.textContent = 'Pause capture';
    pauseBanner.classList.remove('visible');
    statsSection.classList.remove('dimmed');
  }
}

// === Quick-Block Button (D-03) ===
blockBtn.addEventListener('click', async () => {
  const domain = blockBtn.dataset.domain;
  if (!domain) return;

  const added = await addToBlocklist(domain);
  if (added) {
    showToast(`Blocked: ${domain}`, 'success');
  } else {
    showToast(`Already blocked: ${domain}`, 'info');
  }
  blockBtn.disabled = true;
});

// === Edit Blocked Sites Link ===
// D-10: No options page in Phase 1. Show informational toast.
editLink.addEventListener('click', (e) => {
  e.preventDefault();
  showToast('Edit blocklist.json in your extension storage', 'info');
});

// === Toast Notification ===
function showToast(message: string, type: 'success' | 'info'): void {
  toast.textContent = message;
  toast.className = `toast ${type}`;
  setTimeout(() => {
    toast.className = 'toast';
  }, 3000);
}

// === Time Formatting ===
function formatTimeAgo(elapsedMs: number): string {
  const seconds = Math.floor(elapsedMs / 1000);
  if (seconds < 60) return 'just now';
  const minutes = Math.floor(seconds / 60);
  if (minutes < 60) return `${minutes} min ago`;
  const hours = Math.floor(minutes / 60);
  if (hours < 24) return `${hours}h ago`;
  return 'yesterday';
}

// === Init ===
initPopup().catch(console.error);

// Export for testing
export { initPopup, updatePauseUI, formatTimeAgo, showToast };
```
