**Implementation:**

```typescript
import browser from 'webextension-polyfill';
import { BACKFILL_MAX_LOOKBACK_DAYS, BACKFILL_MAX_RESULTS, type CaptureEntry } from './types';
import { isBlocked } from './blocklist';
import { loadStorage, saveStorage, getToday } from './storage';

// Gap threshold: 5 minutes. Shorter gaps are normal (service worker sleep).
const GAP_THRESHOLD_MS = 5 * 60 * 1000;

/**
 * Detect if there's a gap in capture data (D-11).
 * Compares lastCaptureTimestamp against current time.
 * Returns the gap start timestamp, or null if no gap detected.
 */
export async function detectGap(): Promise<number | null> {
  const storage = await loadStorage();
  const lastCapture = storage.lastCaptureTimestamp;

  // First run -- no baseline to compare against
  if (lastCapture === 0) return null;

  const elapsed = Date.now() - lastCapture;

  // Gap must be > 5 minutes to trigger backfill
  // (shorter gaps are normal service worker sleep cycles)
  if (elapsed <= GAP_THRESHOLD_MS) return null;

  return lastCapture;
}

/**
 * Backfill browsing history from chrome.history API for a detected gap (D-11).
 * D-14 (Claude's discretion): Lookback window is 7 days max.
 * Rationale: 7 days balances completeness vs performance. Most gaps are
 * overnight or weekend -- 7 days covers a full vacation scenario.
 * All backfilled entries have source: 'backfill' (D-12).
 */
export async function backfillHistory(gapStartTimestamp: number): Promise<number> {
  const storage = await loadStorage();
  const blocklist = storage.blocklist || [];

  // D-14: Cap lookback at BACKFILL_MAX_LOOKBACK_DAYS (7 days)
  const maxLookback = Date.now() - (BACKFILL_MAX_LOOKBACK_DAYS * 24 * 60 * 60 * 1000);
  const startTime = Math.max(gapStartTimestamp, maxLookback);

  // Query browser history for the gap period
  const historyItems = await browser.history.search({
    text: '',
    startTime,
    endTime: Date.now(),
    maxResults: BACKFILL_MAX_RESULTS,
  });

  let backfillCount = 0;
  const captures = storage.captures || {};

  for (const item of historyItems) {
    // Skip items without URL
    if (!item.url) continue;

    // Filter out non-http URLs
    if (!item.url.startsWith('http://') && !item.url.startsWith('https://')) continue;

    let hostname: string;
    try {
      hostname = new URL(item.url).hostname;
    } catch {
      continue; // Skip malformed URLs
    }

    // Check blocklist
    if (isBlocked(hostname, blocklist)) continue;

    // Determine the date for this history item
    const date = item.lastVisitTime
      ? new Date(item.lastVisitTime).toISOString().split('T')[0]
      : getToday();

    if (!captures[date]) captures[date] = [];

    // D-05: Deduplicate by URL within same day
    const exists = captures[date].some((c: CaptureEntry) => c.url === item.url);
    if (exists) continue;

    // D-12: Mark as backfilled
    const entry: CaptureEntry = {
      url: item.url,
      title: item.title || hostname,
      domain: hostname,
      timestamp: item.lastVisitTime || Date.now(),
      source: 'backfill',
    };

    captures[date].push(entry);
    backfillCount++;
  }

  // Update storage with backfilled captures and new lastCaptureTimestamp
  await saveStorage({
    captures,
    lastCaptureTimestamp: Date.now(),
  });

  console.log(`Second Brain Capture: backfilled ${backfillCount} entries`);
  return backfillCount;
}
```
