```typescript
// Pattern for env var override — config/reader.ts should follow same pattern
function getDataDir(): string {
  const envDir = process.env.SECOND_BRAIN_DATA_DIR;
  if (envDir) {
    const resolved = path.resolve(envDir);
    const homeDir = os.homedir();
    const tmpDir = os.tmpdir();
    if (!resolved.startsWith(homeDir) && !resolved.startsWith(tmpDir)) {
      throw new Error(`Invalid SECOND_BRAIN_DATA_DIR: ...`);
    }
    return resolved;
  }
  return path.join(os.homedir(), '.second-brain');
}
```

    ```typescript
    import { z } from 'zod';
    import { promises as fs } from 'fs';
    import path from 'path';
    import os from 'os';

    export const ConfigSchema = z.object({
      outputDir: z.string()
        .refine(p => path.isAbsolute(p), { message: 'outputDir must be an absolute path' })
        .refine(p => !p.includes('..'), { message: 'outputDir must not contain ..' })
        .optional(),
    });

    export type Config = z.infer<typeof ConfigSchema>;
    ```

    - On any error (file not found, invalid JSON, schema failure), return `{}` (empty defaults)
    - Log warning to stderr on schema validation failure (not silent)

    - Return `config.outputDir || path.join(os.homedir(), 'Documents', 'Obsidian', 'second-brain')`
